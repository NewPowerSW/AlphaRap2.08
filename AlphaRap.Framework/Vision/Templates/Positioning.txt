using System;
using Cognex.VisionPro;
using Cognex.VisionPro.CalibFix;
using Cognex.VisionPro.PMAlign;
using Cognex.VisionPro.ToolBlock;
using Cognex.VisionPro.ToolGroup;

public class CogToolBlockAdvancedScript : CogToolBlockAdvancedScriptBase
{
    private CogToolBlock block;
    /// <summary>绑定原生数据连接的主工具块，脚本只控制执行顺序。</summary>
    public override void Initialize(CogToolGroup host) { base.Initialize(host); block = (CogToolBlock)host; }
    /// <summary>执行采集、定位和结果转换，或独立计算旋转样本。</summary>
    public override bool GroupRun(ref string message, ref CogToolResultConstants result)
    {
        ClearPosition();
        try
        {
            if (block.Tools.Contains("RotationCenter") && Convert.ToBoolean(block.Inputs["RunRotation"].Value))
            {
                block.RunTool(block.Tools["RotationCenter"], ref message, ref result);
                return false;
            }
            CogToolBlock capture = (CogToolBlock)block.Tools["AcquireImage"];
            block.RunTool(capture, ref message, ref result);
            if (capture.RunStatus.Result != CogToolResultConstants.Accept) return false;
            CogCalibNPointToNPointTool calibration = (CogCalibNPointToNPointTool)block.Tools["Calibration"];
            if (Convert.ToBoolean(block.Inputs["UseCalibration"].Value) && calibration.Calibration.Calibrated)
            {
                block.RunTool(calibration, ref message, ref result);
                if (calibration.RunStatus.Result != CogToolResultConstants.Accept) return false;
            }
            CogPMAlignTool pattern = (CogPMAlignTool)block.Tools["LocatePattern"];
            pattern.InputImage.SelectedSpaceName = pattern.InputImage.CoordinateSpaceTree.RootName;
            if (!pattern.Pattern.Trained) throw new InvalidOperationException(Text("请打开 LocatePattern，选择训练图像并框选特征后训练模板。", "Open LocatePattern, select a training image and region, then train the pattern.", "Abra LocatePattern, seleccione imagen y región de entrenamiento y entrene el patrón."));
            block.RunTool(pattern, ref message, ref result);
            if (pattern.RunStatus.Result != CogToolResultConstants.Accept) return false;
            block.RunTool(block.Tools["PositionResult"], ref message, ref result);
        }
        catch (Exception error) { result = CogToolResultConstants.Error; message = error.Message; }
        return false;
    }
    /// <summary>重置结果节点，连接自动清除主输出，不手工转发工具数据。</summary>
    private void ClearPosition()
    {
        block.Outputs["Accepted"].Value = false; block.Outputs["Calibrated"].Value = false;
        foreach (string name in new[] { "x", "y", "A" }) block.Outputs[name].Value = double.NaN;
        block.Outputs["Score"].Value = 0.0; block.Outputs["MatchCount"].Value = 0;
        block.Outputs["OutputImage"].Value = null;
        if (block.Outputs.Contains("RotationValid")) block.Outputs["RotationValid"].Value = false;
        CogToolBlock output = (CogToolBlock)block.Tools["PositionResult"];
        output.Outputs["Accepted"].Value = false; output.Outputs["Calibrated"].Value = false;
        foreach (string name in new[] { "x", "y", "A" }) output.Outputs[name].Value = double.NaN;
        output.Outputs["Score"].Value = 0.0; output.Outputs["MatchCount"].Value = 0;
    }
    /// <summary>使用现场语言包及独立运行的三语底稿。</summary>
    private string Text(string chinese, string english, string spanish)
    {
        int language = Convert.ToInt32(block.Inputs["Language"].Value);
        if (block.UserData.Contains("MessageLanguage") && Convert.ToInt32(block.UserData["MessageLanguage"]) == language && block.UserData.Contains("Message." + english)) return Convert.ToString(block.UserData["Message." + english]);
        return language == 0 ? chinese : language == 1 ? english : spanish;
    }
}
